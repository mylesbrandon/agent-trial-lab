def strategy_returns(prices, signals, transaction_cost=0.0):
"""
Calculate strategy returns from prices and desired positions.
"""
from numbers import Real

```
if len(prices) != len(signals):
    raise ValueError("prices and signals must have equal length")

if not isinstance(transaction_cost, Real) or transaction_cost < 0:
    raise ValueError("transaction_cost cannot be negative")

for price in prices:
    if not isinstance(price, Real) or price <= 0:
        raise ValueError("prices must be positive numbers")

for signal in signals:
    if not isinstance(signal, Real):
        raise ValueError("signals must be numeric")

if not prices:
    return []

results = [0.0]

for i in range(1, len(prices)):
    asset_return = prices[i] / prices[i - 1] - 1
    strategy_return = signals[i - 1] * asset_return

    if i >= 2:
        turnover = abs(signals[i - 1] - signals[i - 2])
        strategy_return -= transaction_cost * turnover

    results.append(strategy_return)

return results
```
